-- =============================================
-- Author: Equipo Datos
-- Create date: 2024-11-02
-- Description: Lista los clientes activos
-- =============================================
CREATE PROCEDURE dbo.usp_Cliente_Listar
@Activo BIT = 1
AS
BEGIN
SET NOCOUNT ON;
SELECT ClienteId, Nombre, Correo, FechaAlta
FROM dbo.Cliente
WHERE Activo = @Activo
ORDER BY Nombre;
END
GO

CREATE PROCEDURE dbo.usp_Factura_Anular
@FacturaId INT,
@Motivo NVARCHAR(200)
AS
BEGIN
SET NOCOUNT ON;
UPDATE dbo.Factura
SET Estado = 'ANULADA', MotivoAnulacion = @Motivo
WHERE FacturaId = @FacturaId;
END
GO
